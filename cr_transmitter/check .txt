/***********************************************************************
 * SX1278 RAW SPI PROBE  (no RadioLib)
 * Single DATA radio only, ESP32 VSPI.
 *
 * Goal: read the version register (0x42) directly and watch every step.
 *   - 0x12 => genuine SX1276/77/78 detected. Wiring + power are GOOD.
 *   - 0x00 => MISO stuck LOW  (bad MISO wire, chip unpowered, wrong NSS)
 *   - 0xFF => MISO stuck HIGH (MISO floating / not connected)
 *   - 0x22 => SX1272/73
 *   - anything else / unstable => noisy wiring, bad solder, weak 3V3
 ***********************************************************************/

#include <SPI.h>

//======================= PINS (DATA radio) ============================
#define SPI_SCK    18
#define SPI_MISO   19
#define SPI_MOSI   23
#define DATA_NSS    5
#define DATA_RST   14
#define DATA_DIO0  26
//======================= SX127x REGISTERS ============================
#define REG_FIFO            0x00
#define REG_OP_MODE         0x01
#define REG_FRF_MSB         0x06
#define REG_FRF_MID         0x07
#define REG_FRF_LSB         0x08
#define REG_VERSION         0x42

#define WNR_WRITE           0x80   // bit7=1 => write, bit7=0 => read

SPIClass sharedSPI(VSPI);
// 2 MHz is slow & safe for probing; raise later once it works.
SPISettings spiSettings(2000000, MSBFIRST, SPI_MODE0);

//======================= LOW-LEVEL SPI ==============================
uint8_t spiReadRegister(uint8_t reg)
{
    sharedSPI.beginTransaction(spiSettings);
    digitalWrite(DATA_NSS, LOW);
    sharedSPI.transfer(reg & 0x7F);            // read: bit7 = 0
    uint8_t value = sharedSPI.transfer(0x00);
    digitalWrite(DATA_NSS, HIGH);
    sharedSPI.endTransaction();
    return value;
}

void spiWriteRegister(uint8_t reg, uint8_t value)
{
    sharedSPI.beginTransaction(spiSettings);
    digitalWrite(DATA_NSS, LOW);
    sharedSPI.transfer(reg | WNR_WRITE);       // write: bit7 = 1
    sharedSPI.transfer(value);
    digitalWrite(DATA_NSS, HIGH);
    sharedSPI.endTransaction();
}

//======================= HARDWARE RESET =============================
void resetRadio()
{
    Serial.println("[RST] Pulsing reset line...");
    pinMode(DATA_RST, OUTPUT);
    digitalWrite(DATA_RST, LOW);    // active low
    delay(10);
    digitalWrite(DATA_RST, HIGH);
    delay(10);                      // datasheet: 5ms min after reset
    Serial.println("[RST] Reset done.");
}

//======================= HELPERS ===================================
void printHex(const char *label, uint8_t v)
{
    Serial.print(label);
    Serial.print("0x");
    if (v < 0x10) Serial.print('0');
    Serial.println(v, HEX);
}

void interpretVersion(uint8_t v)
{
    Serial.println("-----------------------------------------");
    switch (v)
    {
        case 0x12:
            Serial.println(">> 0x12: SX1276/77/78 FOUND. SPI + power OK.");
            break;
        case 0x22:
            Serial.println(">> 0x22: SX1272/73 detected (not 1278).");
            break;
        case 0x00:
            Serial.println(">> 0x00: MISO stuck LOW.");
            Serial.println("   Likely: MISO not connected, chip unpowered,");
            Serial.println("   NSS wrong, or GND missing.");
            break;
        case 0xFF:
            Serial.println(">> 0xFF: MISO stuck HIGH / floating.");
            Serial.println("   Likely: MISO wire open, module not powered.");
            break;
        default:
            Serial.println(">> Unexpected value. Noisy bus / bad solder /");
            Serial.println("   weak 3V3 / wrong SPI mode or speed.");
            break;
    }
    Serial.println("-----------------------------------------");
}

//======================= SETUP =====================================
void setup()
{
    Serial.begin(115200);
    delay(1000);
    Serial.println();
    Serial.println("=========================================");
    Serial.println(" SX1278 RAW SPI PROBE (no RadioLib)");
    Serial.println("=========================================");

    // NSS idle HIGH before anything else.
    pinMode(DATA_NSS, OUTPUT);
    digitalWrite(DATA_NSS, HIGH);

    pinMode(DATA_DIO0, INPUT);

    Serial.print("Pins  SCK="); Serial.print(SPI_SCK);
    Serial.print(" MISO=");     Serial.print(SPI_MISO);
    Serial.print(" MOSI=");     Serial.print(SPI_MOSI);
    Serial.print(" NSS=");      Serial.print(DATA_NSS);
    Serial.print(" RST=");      Serial.print(DATA_RST);
    Serial.print(" DIO0=");     Serial.println(DATA_DIO0);

    sharedSPI.begin(SPI_SCK, SPI_MISO, SPI_MOSI, -1); // we drive NSS manually
    Serial.println("[SPI] Bus started (2 MHz, MODE0).");

    resetRadio();

    // Read version several times to check stability.
    Serial.println();
    Serial.println("[VER] Reading REG_VERSION (0x42) x5:");
    uint8_t v = 0;
    for (int i = 0; i < 5; i++)
    {
        v = spiReadRegister(REG_VERSION);
        Serial.print("  attempt "); Serial.print(i + 1);
        printHex(" -> ", v);
        delay(50);
    }
    interpretVersion(v);

    // Dump a few registers to eyeball the bus.
    Serial.println();
    Serial.println("[DUMP] Key registers after reset:");
    printHex("  REG_OP_MODE (0x01) = ", spiReadRegister(REG_OP_MODE));
    printHex("  REG_FRF_MSB (0x06) = ", spiReadRegister(REG_FRF_MSB));
    printHex("  REG_FRF_MID (0x07) = ", spiReadRegister(REG_FRF_MID));
    printHex("  REG_FRF_LSB (0x08) = ", spiReadRegister(REG_FRF_LSB));

    // WRITE/READ-BACK test: proves MOSI + latching, not just MISO.
    Serial.println();
    Serial.println("[RW] Write/read-back test on REG_OP_MODE:");
    Serial.println("     (put chip in SLEEP + LoRa = 0x80)");
    spiWriteRegister(REG_OP_MODE, 0x80);   // LoRa sleep
    delay(10);
    uint8_t back = spiReadRegister(REG_OP_MODE);
    printHex("  read back = ", back);
    if (back == 0x80)
        Serial.println("  >> WRITE PATH OK (MOSI + NSS latching good).");
    else
        Serial.println("  >> WRITE FAILED: MOSI issue, or chip not responding.");

    Serial.println();
    Serial.println("Probe complete. See interpretation above.");
}

void loop()
{
    // Re-read version once a second so you can wiggle wires and watch.
    uint8_t v = spiReadRegister(REG_VERSION);
    printHex("[LOOP] VERSION = ", v);
    delay(1000);
}
